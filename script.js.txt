function sendMessage() {
    const input = document.getElementById('chatInput');
    const text = input.value.trim().toLowerCase();
    if (!text) return;
    const chatBody = document.getElementById('chatBody');
    chatBody.innerHTML += `<div class="chat-msg user-msg">${input.value}</div>`;
    input.value = '';
    chatBody.scrollTop = chatBody.scrollHeight;
    setTimeout(() => {
        let r = "I can answer questions about our IT, Development, Cloud, and Cybersecurity services.";
        if (text.includes('erp')) r = "Our Custom ERP solutions integrate finance, HR, supply chain, and operations seamlessly. Book a consultation to discuss your needs!";
        else if (text.includes('cyber') || text.includes('security')) r = "We provide elite Cyber Security: Penetration testing, SOC, and Active Directory assessments.";
        else if (text.includes('cloud')) r = "Our Cloud services ensure scalable infrastructure, cost optimization, and MDM/MAM solutions.";
        else if (text.includes('store') || text.includes('laptop')) r = "Check out our Store section for HP, Dell, and Lenovo laptops. You can inquire directly via WhatsApp!";
        chatBody.innerHTML += `<div class="chat-msg bot-msg">${r}</div>`;
        chatBody.scrollTop = chatBody.scrollHeight;
    }, 600);
}

const mCanvas = document.getElementById('matrix-canvas');
const mCtx = mCanvas.getContext('2d');
mCanvas.width = window.innerWidth; mCanvas.height = window.innerHeight;
let drops = Array(Math.floor(mCanvas.width / 14)).fill(1);
function drawMatrix() {
    mCtx.fillStyle = 'rgba(3, 3, 5, 0.1)';
    mCtx.fillRect(0, 0, mCanvas.width, mCanvas.height);
    mCtx.fillStyle = '#00F0FF'; mCtx.font = '14px monospace';
    for (let i = 0; i < drops.length; i++) {
        const t = "0123456789</>{}"[(Math.random() * 16) | 0];
        mCtx.fillText(t, i * 14, drops[i] * 14);
        if (drops[i] * 14 > mCanvas.height && Math.random() > 0.975) drops[i] = 0;
        drops[i]++;
    }
}

// --- 3D Shape Morphing AI Core ---
const canvas = document.getElementById('ai-canvas');
const ctx = canvas.getContext('2d');
let w, h, cx, cy, radius;

function resize() {
    w = canvas.width = window.innerWidth;
    h = canvas.height = window.innerHeight;
    cx = w / 2; cy = h / 2;
    radius = Math.min(w, h) * 0.3;
}
resize();

const N = 180;
let currentShape = [];
let targetShape = [];
let morphProgress = 1;
let restTimer = 0;
let rotX = 0, rotY = 0;
let dataPackets = [];

const palettes = [
    ['#00F0FF', '#FF0055', '#FF7700'], 
    ['#FF0055', '#FF7700', '#00F0FF'], 
    ['#39FF14', '#00F0FF', '#BF00FF'], 
    ['#BF00FF', '#FF0055', '#39FF14']  
];
let currentColors = palettes[0];
let targetColors = palettes[1];
let colorProgress = 1;

function generateSphere() {
    let pts = []; let lat = 9, lon = 20;
    for (let i = 0; i < lat; i++) {
        for (let j = 0; j < lon; j++) {
            let theta = (i / lat) * Math.PI; let phi = (j / lon) * 2 * Math.PI;
            pts.push({ x: radius * Math.sin(theta) * Math.cos(phi), y: radius * Math.cos(theta), z: radius * Math.sin(theta) * Math.sin(phi) });
        }
    }
    return pts;
}

function generateInfinity() {
    let pts = []; let p = 2, q = 3; let R = radius * 0.8, r = radius * 0.3;
    for (let i = 0; i < N; i++) {
        let t = (i / N) * 2 * Math.PI;
        pts.push({ x: (R + r * Math.cos(q * t)) * Math.cos(p * t), y: (R + r * Math.cos(q * t)) * Math.sin(p * t), z: r * Math.sin(q * t) });
    }
    return pts;
}

function generatePolygons() {
    let pts = []; let layers = 6; let pointsPerLayer = N / layers;
    for (let i = 0; i < layers; i++) {
        let y = radius * ((i / (layers - 1)) * 2 - 1);
        let currentRadius = radius * Math.sin((i / (layers - 1)) * Math.PI);
        for (let j = 0; j < pointsPerLayer; j++) {
            let angle = (j / pointsPerLayer) * 2 * Math.PI;
            pts.push({ x: currentRadius * Math.cos(angle), y: y, z: currentRadius * Math.sin(angle) });
        }
    }
    return pts;
}

function generateRings() {
    let pts = []; let ringPoints = N / 3;
    for (let i = 0; i < N; i++) {
        let ring = Math.floor(i / ringPoints);
        let t = ((i % ringPoints) / ringPoints) * 2 * Math.PI;
        if (ring === 0) pts.push({ x: radius * Math.cos(t), y: radius * Math.sin(t), z: 0 });
        else if (ring === 1) pts.push({ x: radius * Math.cos(t), y: 0, z: radius * Math.sin(t) });
        else pts.push({ x: 0, y: radius * Math.cos(t), z: radius * Math.sin(t) });
    }
    return pts;
}

function getRandomShape() {
    const shapes = [generateSphere, generateInfinity, generatePolygons, generateRings];
    return shapes[Math.floor(Math.random() * shapes.length)]();
}

currentShape = generateSphere();
targetShape = generateInfinity();

let connections = [];
let lat = 9, lon = 20;
for (let i = 0; i < lat; i++) {
    for (let j = 0; j < lon; j++) {
        let p1 = i * lon + j;
        let p2 = i * lon + (j + 1) % lon;
        let p3 = (i + 1) * lon + j;
        if (i < lat - 1) connections.push([p1, p3]);
        connections.push([p1, p2]);
    }
}

function hexToRgb(hex) {
    let r = parseInt(hex.slice(1, 3), 16); let g = parseInt(hex.slice(3, 5), 16); let b = parseInt(hex.slice(5, 7), 16);
    return [r, g, b];
}
function lerp(a, b, t) { return a + (b - a) * t; }
function lerpColor(c1, c2, t) {
    let rgb1 = hexToRgb(c1); let rgb2 = hexToRgb(c2);
    let r = Math.round(lerp(rgb1[0], rgb2[0], t)); let g = Math.round(lerp(rgb1[1], rgb2[1], t)); let b = Math.round(lerp(rgb1[2], rgb2[2], t));
    return `rgb(${r},${g},${b})`;
}
function lerpPoint(p1, p2, t) {
    return { x: lerp(p1.x, p2.x, t), y: lerp(p1.y, p2.y, t), z: lerp(p1.z, p2.z, t) };
}

function drawAI() {
    ctx.clearRect(0, 0, w, h);
    rotX += 0.003; rotY += 0.005;

    if (morphProgress < 1) {
        morphProgress += 0.008;
        if (morphProgress >= 1) { morphProgress = 1; restTimer = 0; }
    } else {
        restTimer++;
        if (restTimer > 180) {
            morphProgress = 0;
            currentShape = targetShape.map(p => ({ ...p }));
            targetShape = getRandomShape();
            currentColors = [...targetColors];
            let nextColorIndex = (palettes.indexOf(targetColors) + 1) % palettes.length;
            targetColors = palettes[nextColorIndex];
            colorProgress = 0;
        }
    }

    if (colorProgress < 1) colorProgress += 0.005;

    let displayPoints = [];
    for (let i = 0; i < N; i++) {
        displayPoints.push(lerpPoint(currentShape[i], targetShape[i], morphProgress));
    }

    const projected = displayPoints.map(p => {
        let cosY = Math.cos(rotY), sinY = Math.sin(rotY);
        let x1 = p.x * cosY - p.z * sinY; let z1 = p.x * sinY + p.z * cosY;
        let cosX = Math.cos(rotX), sinX = Math.sin(rotX);
        let y1 = p.y * cosX - z1 * sinX; let z2 = p.y * sinX + z1 * cosX;
        let scale = 400 / (400 + z2);
        return { x: cx + x1 * scale, y: cy + y1 * scale, z: z2, scale: scale };
    });

    let lineColor = lerpColor(currentColors[0], targetColors[0], colorProgress);
    let nodeColor1 = lerpColor(currentColors[1], targetColors[1], colorProgress);
    let nodeColor2 = lerpColor(currentColors[2], targetColors[2], colorProgress);

    connections.forEach((c, idx) => {
        let p1 = projected[c[0]]; let p2 = projected[c[1]];
        let opacity = Math.min(p1.scale, p2.scale) * 0.6;
        ctx.strokeStyle = lineColor.replace('rgb', 'rgba').replace(')', `, ${opacity * 0.4})`);
        ctx.lineWidth = 1;
        ctx.beginPath(); ctx.moveTo(p1.x, p1.y); ctx.lineTo(p2.x, p2.y); ctx.stroke();
    });

    projected.forEach((p, idx) => {
        let color = idx % 2 === 0 ? nodeColor1 : nodeColor2;
        let rgb = color.match(/\d+/g);
        ctx.fillStyle = `rgba(${rgb[0]}, ${rgb[1]}, ${rgb[2]}, ${p.scale * 0.9})`;
        ctx.shadowBlur = 10; ctx.shadowColor = color;
        ctx.beginPath(); ctx.arc(p.x, p.y, p.scale * 2.5, 0, Math.PI * 2); ctx.fill();
    });
    ctx.shadowBlur = 0;

    if (Math.random() > 0.92) {
        let c = connections[Math.floor(Math.random() * connections.length)];
        dataPackets.push({ p1: c[0], p2: c[1], t: 0, color: currentColors[Math.floor(Math.random()*3)] });
    }

    dataPackets.forEach((d, index) => {
        d.t += 0.05;
        if (d.t > 1) { dataPackets.splice(index, 1); return; }
        let p1 = projected[d.p1]; let p2 = projected[d.p2];
        let x = p1.x + (p2.x - p1.x) * d.t; let y = p1.y + (p2.y - p1.y) * d.t;
        ctx.fillStyle = d.color; ctx.shadowBlur = 15; ctx.shadowColor = d.color;
        ctx.beginPath(); ctx.arc(x, y, 3, 0, Math.PI * 2); ctx.fill();
    });
    ctx.shadowBlur = 0;
}

function loop() {
    drawMatrix(); drawAI(); requestAnimationFrame(loop);
}
